export const updateUser = async (userID, updateData) => {
  if (!userID) throw new Error("No userID provided");

  console.log("========== STRICT UPDATE USER ==========");
  console.log("userID:", userID);
  console.log("updateData:", updateData);

  const userRef = db.collection("users").doc(userID);
  const docSnap = await userRef.get();

  if (!docSnap.exists) {
    throw new Error("User document not found");
  }

  const existingData = docSnap.data();
  console.log("existing user fields:", Object.keys(existingData));

  const update = {};
  const rejectedFields = [];

  // STEP 1: validate against existing schema
  for (const key of Object.keys(updateData)) {
    if (!(key in existingData)) {
      rejectedFields.push(key);
      continue; // ❌ skip non-existing fields
    }

    const newValue = updateData[key];
    const oldValue = existingData[key];

    const isObject =
      typeof newValue === "object" &&
      newValue !== null &&
      !Array.isArray(newValue) &&
      typeof oldValue === "object" &&
      oldValue !== null;

    if (isObject) {
      update[key] = { ...oldValue, ...newValue };
    } else {
      update[key] = newValue;
    }
  }

  console.log("valid update payload:", update);
  console.log("rejected fields:", rejectedFields);

  // STEP 2: if nothing valid → stop
  if (Object.keys(update).length === 0) {
    throw new Error("No valid existing fields to update");
  }

  // STEP 3: update Firestore
  await userRef.update(update);

  const updatedSnap = await userRef.get();

  console.log("update complete");
  console.log("========== END STRICT UPDATE ==========");

  return {
    data: updatedSnap.data(),
    rejectedFields,
  };
};
